import json
import os

from config import DATABASE_FILE


def load_games():
    if not os.path.exists(DATABASE_FILE):
        return {}

    try:
        with open(DATABASE_FILE, "r", encoding="utf-8") as f:
            return json.load(f)
    except (json.JSONDecodeError, FileNotFoundError):
        return {}


def save_games(data):
    with open(DATABASE_FILE, "w", encoding="utf-8") as f:
        json.dump(data, f, indent=4, ensure_ascii=False)


def add_game(name, status, event, event_key, link):
    games = load_games()

    games[name] = {
        "status": status,
        "event": event,
        "event_key": event_key,
        "link": link
    }

    save_games(games)


def edit_game(name, status, event, event_key, link):
    games = load_games()

    if name not in games:
        return False

    games[name]["status"] = status
    games[name]["event"] = event
    games[name]["event_key"] = event_key
    games[name]["link"] = link

    save_games(games)
    return True


def delete_game(name):
    games = load_games()

    if name in games:
        del games[name]
        save_games(games)
        return True

    return False


def get_game(name):
    games = load_games()
    return games.get(name)


def get_all_games():
    return load_games()
